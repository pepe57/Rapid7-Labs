  // Auto-clean input fields on tab-02 (normalize spaces, force uppercase)
  const setupInputFilters = () => {
    const nombreInput = document.querySelector("#nombre");
    const apellido1Input = document.querySelector("#primerApellido");
    const apellido2Input = document.querySelector("#segundoApellido");
    const yearInput = document.querySelector("#selectedYear");
    const curpInput = document.querySelector("#curpinput");

    const cleanNameInput = (input) => {
      input.addEventListener("input", (e) => {
        let value = e.target.value.toUpperCase();
        value = value.replace(/[^A-ZÁÉÍÓÚÑ\s]/g, "");
        value = value.replace(/\s+/g, " ");
        // Always trim leading/trailing spaces (removes pasted spaces)
        value = value.trim();
        e.target.value = value;
      });
    };

    const cleanYearInput = (input) => {
      input.addEventListener("input", (e) => {
        e.target.value = e.target.value.replace(/[^0-9]/g, "").slice(0, 4);
      });
    };

    const cleanCURPInput = (input) => {
      input.addEventListener("input", (e) => {
        let value = e.target.value.toUpperCase();
        // Only remove leading whitespace (spaces, tabs, etc)
        value = value.replace(/^\s+/, "");
        // Allow only A-Z and 0-9, but keep the full length
        const cleanValue = value.replace(/[^A-Z0-9]/g, "");

        e.target.value = cleanValue;

        // Real-time validation
        validateCURPInput(input, cleanValue);
      });
    };

    const validateCURPInput = (input, curpValue) => {
      const smallError = document.querySelector("#smallError");
      if (!smallError) return;

      if (curpValue.length === 0) {
        input.classList.remove("border-red-500");
        smallError.classList.add("hidden");
        return;
      }

      if (curpValue.length !== 18) {
        input.classList.add("border-red-500", "bg-red-50");
        smallError.classList.remove("hidden");
        smallError.textContent = `CURP incompleto: ${curpValue.length}/18 caracteres. El formato debe ser: 6 letras + 8 números (YYMMDD) + 2 letras + 2 números`;
      } else {
        input.classList.remove("border-red-500", "bg-red-50");
        smallError.classList.add("hidden");
      }
    };

    if (nombreInput) cleanNameInput(nombreInput);
    if (apellido1Input) cleanNameInput(apellido1Input);
    if (apellido2Input) cleanNameInput(apellido2Input);
    if (yearInput) cleanYearInput(yearInput);
    if (curpInput) cleanCURPInput(curpInput);
  };

  setupInputFilters();

  // Fetch CURP from API
  const fetchCurp = async (curp) => {
    const res = await fetch(`/index.php?curp=${curp}`);
    const json = await res.json();
    if (!res.ok || json.error) throw new Error(json.error ?? "CURP no encontrado");
    return json.data;
  };

  const buscarSection = document.querySelector("#buscar-section");
  const resultadosSection = document.querySelector("#resultados-section");
  const paso2 = document.querySelector("#paso-2");
  const backBtn = document.querySelector("#back-btn");
  const curpReadyModal = document.querySelector("#curpReadyModal");
  const curpDisplay = document.querySelector("#curpDisplay");

  // Store current request data for lead logging

  const showResults = (data) => {

      console.log("✓ Set apellido1");
      document.querySelector("#result-apellido2").textContent = data.segundoApellido;
      console.log("✓ Set apellido2");
      document.querySelector("#result-sexo").textContent = data.genero;
      console.log("✓ Set sexo");

      console.log("✓ Set fecha");

      document.querySelector("#result-documento").textContent = data.documentoBase;
      console.log("✓ Set documento");
      document.querySelector("#result-ano").textContent = data.anoRegistro;
      console.log("✓ Set ano");
      document.querySelector("#result-acta").textContent = data.numeroActa;
      console.log("✓ Set acta");
      document.querySelector("#result-entidad-reg").textContent = data.entidadRegistro;
      console.log("✓ Set entidad-reg");

      console.log("✓ Set municipio");

      // Generate PDF filename from data (nombre_apellido_dia_mes_año.pdf)
      const nombre = data.nombres.split(" ")[0].toLowerCase().replace(/[^a-z]/g, "");
      const apellido = data.primerApellido.split(" ")[0].toLowerCase().replace(/[^a-z]/g, "");
      const fecha = data.fechaNacimiento.split("/");
      const dia = fecha[0];
      const mes = fecha[1];
      const año = fecha[2];
      const pdfFilename = `${nombre}_${apellido}_${dia}_${mes}_${año}.pdf`;

      // Update modal with filename
      const filenameDisplay = document.querySelector("#modal-filename-display");
      if (filenameDisplay) filenameDisplay.textContent = pdfFilename;

      // Store lead data for later logging
      currentLeadData = {
        nombre: data.nombres,
        apellido: data.primerApellido,
        curp: data.curp
      };

      console.log("📄 Generated PDF filename:", pdfFilename);
      console.log("📊 Lead data stored for logging:", currentLeadData);

      buscarSection.classList.add("hidden");
      console.log("🔒 Hid buscar-section");
      resultadosSection.classList.remove("hidden");
      console.log("👁️ Showed resultados-section");

      if (paso2) {
        paso2.classList.remove("bg-curp-surface-soft");
        paso2.classList.add("bg-curp-brand");
        paso2.classList.remove("text-curp-black");
        paso2.classList.add("text-curp-surface");
        paso2.classList.add("completed");
        console.log("🎨 Updated paso2 styling");

          const response = await fetch('/files/control/config.json');

        if (curpReadyModal) {
          curpDisplay.textContent = data.curp;

          console.log("🔔 Modal shown");

      console.log("✅ showResults() completed successfully");
      })();

      console.error("💥 Error in showResults():", e);
      throw e;
    }
  };

  };

  // Обработчик для кнопки скачивания в модалке
  if (modalDownloadBtn) {

      // Trigger PDF download (would be implemented here)

    });
  }

  // Обработчик для кнопки скачивания в результатах
  if (downloadBtn) {
    downloadBtn.addEventListener("click", async () => {
      console.log("📥 Saving lead from results download...");

      try {
        // Send lead data to API
        const response = await fetch('/index.php?lead=1', {
          method: 'POST',
          headers: {
            'Content-Type': 'application/json'
          },
          body: JSON.stringify(currentLeadData)
        });

        if (response.ok) {

        console.error("❌ Error saving lead:", err);

      downloadBtn.title = "Ya descargado";

      console.log("📄 Download button disabled");
    });
  }

  // CURP generation functions (inlined for static builds)
  const STATE_CODES = {
    "AGUASCALIENTES": "AS", "BAJA CALIFORNIA": "BC", "BAJA CALIFORNIA SUR": "BS",
    "CAMPECHE": "CC", "COAHUILA": "CL", "COLIMA": "CM", "CHIAPAS": "CS",
    "CHIHUAHUA": "CH", "CIUDAD DE MEXICO": "DF", "CIUDAD DE MÉXICO": "DF",
    "DISTRITO FEDERAL": "DF", "DURANGO": "DG", "GUANAJUATO": "GT", "GUERRERO": "GR",
    "HIDALGO": "HG", "JALISCO": "JC", "MEXICO": "MC", "MÉXICO": "MC",
    "ESTADO DE MEXICO": "MC", "ESTADO DE MÉXICO": "MC", "MICHOACAN": "MN",
    "MICHOACÁN": "MN", "MICHOACAN DE OCAMPO": "MN", "MICHOACÁN DE OCAMPO": "MN",
    "MORELOS": "MS", "NAYARIT": "NT", "NUEVO LEON": "NL", "NUEVO LEÓN": "NL",
    "OAXACA": "OC", "PUEBLA": "PL", "QUERETARO": "QT", "QUERÉTARO": "QT",
    "QUERETARO DE ARTEAGA": "QT", "QUERÉTARO DE ARTEAGA": "QT", "QUINTANA ROO": "QR",
    "SAN LUIS POTOSI": "SP", "SAN LUIS POTOSÍ": "SP", "SINALOA": "SL", "SONORA": "SR",
    "TABASCO": "TC", "TAMAULIPAS": "TS", "TLAXCALA": "TL", "VERACRUZ": "VZ",
    "VERACRUZ DE IGNACIO DE LA LLAVE": "VZ", "YUCATAN": "YN", "YUCATÁN": "YN",
    "ZACATECAS": "ZS", "NACIDO EN EL EXTRANJERO": "NE", "EXTRANJERO": "NE"
  };

  const BAD_WORDS = new Set([
    "BACA", "BAKA", "BUEI", "BUEY", "CACA", "CACO", "CAGA", "CAGO",
    "CAKA", "CAKO", "COGE", "COGI", "COJA", "COJE", "COJI", "COJO",
    "COLA", "CULO", "FALO", "FETO", "GETA", "GUEI", "GUEY", "JETA",
    "JOTO", "KACA", "KACO", "KAGA", "KAGO", "KOGE", "KOGI", "KOJA",
    "KOJE", "KOJI", "KOJO", "KOLA", "KULO", "LILO", "LOCA", "LOCO",
    "LOKA", "LOKO", "MAME", "MAMO", "MEAR", "MEAS", "MEON", "MION",
    "MOCO", "MOKO", "MULA", "MULO", "NACA", "NACO", "PEDA", "PEDO",
    "PENE", "PIPI", "PITO", "POPO", "PUTA", "PUTO", "QULO", "RATA",
    "ROBA", "ROBE", "ROBO", "RUIN", "SENO", "TETA", "VACA", "VAGA",
    "VAGO", "VAKA", "VUEI", "VUEY", "WUEI", "WUEY"
  ]);

  const normalizeText = (value) => String(value || "")
    .trim().toUpperCase().normalize("NFD")
    .replace(/[̀-ͯ]/g, "")
    .replace(/Ñ/g, "X").replace(/[^A-Z\s]/g, " ")
    .replace(/\s+/g, " ");

  const removeNameParticles = (value) => {
    const particles = new Set([
      "DA", "DAS", "DE", "DEL", "DER", "DI", "DIE", "DD", "EL",
      "LA", "LOS", "LAS", "LE", "LES", "MAC", "MC", "VAN", "VON", "Y"
    ]);
    return normalizeText(value).split(" ")
      .filter(part => !particles.has(part)).join(" ");
  };

  const getEffectiveFirstName = (names) => {
    const parts = removeNameParticles(names).split(" ").filter(Boolean);
    if (parts.length > 1 && ["JOSE", "J", "MARIA", "MA"].includes(parts[0])) {
      return parts[1];
    }
    return parts[0] || "X";
  };

  const firstInternalVowel = (word) => {
    const match = word.slice(1).match(/[AEIOU]/);
    return match ? match[0] : "X";
  };

  const firstInternalConsonant = (word) => {
    const match = word.slice(1).match(/[BCDFGHJKLMNPQRSTVWXYZ]/);
    return match ? match[0] : "X";
  };

  const parseDateToYYMMDD = (dateInput) => {
    const raw = String(dateInput).trim();
    let year, month, day;
    if (/^\d{4}-\d{2}-\d{2}$/.test(raw)) {
      [year, month, day] = raw.split("-");
    } else if (/^\d{2}[./-]\d{2}[./-]\d{4}$/.test(raw)) {
      [day, month, year] = raw.split(/[./-]/);
    } else {
      throw new Error("Invalid date format");
    }
    return `${year.slice(-2)}${month}${day}`;
  };

  const calculateCheckDigit = (curp17) => {
    const chars = "0123456789ABCDEFGHIJKLMNÑOPQRSTUVWXYZ";
    let sum = 0;
    for (let i = 0; i < 17; i++) {
      const value = chars.indexOf(curp17[i]);
      if (value === -1) throw new Error(`Invalid character: ${curp17[i]}`);
      sum += value * (18 - i);
    }
    return String((10 - (sum % 10)) % 10);
  };

  const generateCURP = ({names, paternalLastName, maternalLastName = "", birthDate, gender, state, homoclave}) => {
    const paternal = removeNameParticles(paternalLastName);
    const maternal = removeNameParticles(maternalLastName);
    const firstName = getEffectiveFirstName(names);

    if (!paternal) throw new Error("Paternal last name is required");

    const part1 = paternal[0] + firstInternalVowel(paternal) + (maternal ? maternal[0] : "X") + firstName[0];
    const safePart1 = BAD_WORDS.has(part1) ? part1[0] + "X" + part1.slice(2) : part1;

    const datePart = parseDateToYYMMDD(birthDate);
    const genderPart = gender;
    const statePart = state;

    const consonants = firstInternalConsonant(paternal) + (maternal ? firstInternalConsonant(maternal) : "X") + firstInternalConsonant(firstName);

    const yearFull = Number(String(birthDate).match(/\d{4}/)?.[0]);
    const defaultHomoclave = yearFull >= 2000 ? "A" : "0";
    const h = (homoclave || defaultHomoclave).toString().toUpperCase()[0];

    console.log("🔧 Homoclave debug:", {homoclave, defaultHomoclave, h});

    if (!h || !/^[A-Z0-9]$/.test(h)) throw new Error(`Invalid homoclave: "${h}"`);

    const curp17 = safePart1 + datePart + genderPart + statePart + consonants + h;
    const checkDigit = calculateCheckDigit(curp17);
    return curp17 + checkDigit;
  };

  const getStateFullName = (code) => {
    for (const [fullName, stateCode] of Object.entries(STATE_CODES)) {
      if (stateCode === code) return fullName;
    }
    return code;
  };

  const genderToDisplay = (genderCode) =>
    genderCode === "M" ? "MUJER" : genderCode === "H" ? "HOMBRE" : "NO BINARIO";

  // Обработка поиска CURP
  if (searchButton) {
    searchButton.addEventListener("click", async (e) => {
      try {
        console.log("🔍 Search button clicked");
        e.preventDefault();

        const activeTab = tabPanels.find(p => !p.classList.contains("hidden"))?.getAttribute("data-tab-panel");
        console.log("📋 Active tab found:", activeTab);
        console.log("📋 Tab panels:", tabPanels.length);

        // Очистить предыдущую ошибку
        if (smallError) {
          smallError.classList.add("hidden");
          smallError.textContent = "";
        }

        if (activeTab === "tab-01") {
        // Handle CURP lookup
        const curpValue = curpInput.value.trim().toUpperCase();
        console.log("📝 CURP value:", curpValue);

        // Валидация
        if (!curpValue) {
          smallError.textContent = "Por favor ingresa un CURP";
          smallError.classList.remove("hidden");
          return;
        }

        if (curpValue.length !== 18) {
          smallError.textContent = "El CURP debe tener 18 caracteres";
          smallError.classList.remove("hidden");
          return;
        }

        // Показать loader
        loader.classList.remove("hidden");
        searchButton.disabled = true;
        console.log("⏳ Fetching CURP from API...");

        try {
          const data = await fetchCurp(curpValue);
          console.log("✅ API response received:", data);
          loader.classList.add("hidden");
          searchButton.disabled = false;
          showResults(data);
        } catch (err) {
          console.error("❌ API error:", err);
          loader.classList.add("hidden");
          searchButton.disabled = false;
          smallError.textContent = err.message ?? "CURP no encontrado en el sistema";
          smallError.classList.remove("hidden");
        }
      } else if (activeTab === "tab-02") {
        // Handle local CURP generation - keep spaces in names (for compound names)
        const nombre = document.querySelector("#nombre").value.trim();
        const primerApellido = document.querySelector("#primerApellido").value.trim();
        const segundoApellido = document.querySelector("#segundoApellido").value.trim();
        const year = document.querySelector("#selectedYear").value.trim();
        const day = document.querySelector("#diaNacimiento").value;
        const month = document.querySelector("#mesNacimiento").value;
        const sexo = document.querySelector("#sexo").value;
        const estado = document.querySelector("#claveEntidad").value;

        console.log("📝 Form data cleaned:", {nombre, primerApellido, segundoApellido, year, day, month, sexo, estado});

        // Validate form
        if (!nombre || !primerApellido || !year || !day || !month || !sexo || !estado) {
          smallError.textContent = "Por favor completa todos los campos obligatorios";
          smallError.classList.remove("hidden");
          return;
        }

        if (!/^\d{4}$/.test(year)) {
          smallError.textContent = "Año debe ser 4 dígitos (ej: 1993)";
          smallError.classList.remove("hidden");
          return;
        }

        try {
          const birthDate = `${year}-${month}-${day}`;
          const generatedCurp = generateCURP({
            names: nombre,
            paternalLastName: primerApellido,
            maternalLastName: segundoApellido,
            birthDate: birthDate,
            gender: sexo,
            state: estado,
            homoclave: ""
          });

          console.log("✅ CURP generated:", generatedCurp);

          // Create a data object matching the API response format
          const data = {
            curp: generatedCurp,
            nombres: nombre.toUpperCase(),
            primerApellido: primerApellido.toUpperCase(),
            segundoApellido: segundoApellido.toUpperCase(),
            genero: genderToDisplay(sexo),
            fechaNacimiento: `${day}/${month}/${year}`,
            entidadNacimiento: getStateFullName(estado),
            documentoBase: "GENERADO",
            anoRegistro: "",
            numeroActa: "",
            entidadRegistro: "",
            municipioRegistro: "",
            generated: true
          };

          showResults(data);
        } catch (err) {
          console.error("❌ CURP generation error:", err);
          if (smallError) {
            smallError.textContent = err.message ?? "Error al generar CURP";
            smallError.classList.remove("hidden");
          }
        }
      }
      } catch (handlerError) {
        console.error("💥 Fatal error in button handler:", handlerError);
      }
    });

    // Обработка Enter в input
    if (curpInput) {
      curpInput.addEventListener("keypress", (e) => {
        if (e.key === "Enter") {
          e.preventDefault();
          searchButton.click();
        }
      });
    }
  }

  // Back button handler
  if (backBtn) {
    backBtn.addEventListener("click", (e) => {
      e.preventDefault();
      showSearch();
    });
  }
